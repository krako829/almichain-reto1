! Almi-Chain · Router R-ALMI (2911): subinterfaces, DHCP, NAT, ACL y SSH
! Pegar en la pestaña CLI de Packet Tracer bloque a bloque.
! <contraseña> se cambia por la del gestor de contraseñas (KeePassXC). No subir las contraseñas reales.
!
enable
configure terminal
hostname R-ALMI
! Primero la longitud mínima: solo se aplica a las contraseñas que se pongan DESPUÉS
security passwords min-length 12
enable secret <contraseña>
service password-encryption
no ip domain-lookup
banner motd #Acceso solo para personal autorizado de Almi-Chain#

interface g0/0
no shutdown
exit
interface g0/0.10
encapsulation dot1Q 10
ip address 192.168.10.1 255.255.255.0
ip nat inside
exit
interface g0/0.20
encapsulation dot1Q 20
ip address 192.168.20.1 255.255.255.0
ip nat inside
exit
interface g0/0.30
encapsulation dot1Q 30
ip address 192.168.30.1 255.255.255.0
ip nat inside
exit
interface g0/0.99
encapsulation dot1Q 99
ip address 192.168.99.1 255.255.255.0
exit

interface g0/1
description WAN-OPERADOR
ip address 203.0.113.2 255.255.255.252
ip nat outside
no shutdown
exit
ip route 0.0.0.0 0.0.0.0 203.0.113.1

! DHCP solo en el rango del plan (.100-.199 en gestión y TPV, .100-.250 en clientes)
ip dhcp excluded-address 192.168.10.1 192.168.10.99
ip dhcp excluded-address 192.168.10.200 192.168.10.254
ip dhcp excluded-address 192.168.20.1 192.168.20.99
ip dhcp excluded-address 192.168.20.200 192.168.20.254
ip dhcp excluded-address 192.168.30.1 192.168.30.99
ip dhcp excluded-address 192.168.30.251 192.168.30.254
ip dhcp pool GESTION
network 192.168.10.0 255.255.255.0
default-router 192.168.10.1
dns-server 198.51.100.10
exit
ip dhcp pool TPV
network 192.168.20.0 255.255.255.0
default-router 192.168.20.1
dns-server 198.51.100.10
exit
ip dhcp pool CLIENTES
network 192.168.30.0 255.255.255.0
default-router 192.168.30.1
dns-server 198.51.100.10
exit

access-list 1 permit 192.168.10.0 0.0.0.255
access-list 1 permit 192.168.20.0 0.0.0.255
access-list 1 permit 192.168.30.0 0.0.0.255
ip nat inside source list 1 interface g0/1 overload

ip access-list extended CLIENTES-SOLO-INTERNET
deny ip 192.168.30.0 0.0.0.255 192.168.10.0 0.0.0.255
deny ip 192.168.30.0 0.0.0.255 192.168.20.0 0.0.0.255
deny ip 192.168.30.0 0.0.0.255 192.168.99.0 0.0.0.255
permit ip any any
exit
interface g0/0.30
ip access-group CLIENTES-SOLO-INTERNET in
exit

ip access-list extended TPV-SOLO-ODOO
permit tcp 192.168.20.0 0.0.0.255 host 192.168.10.30 eq 443
permit tcp 192.168.20.0 0.0.0.255 host 192.168.10.30 eq www
deny ip 192.168.20.0 0.0.0.255 192.168.10.0 0.0.0.255
deny ip 192.168.20.0 0.0.0.255 192.168.99.0 0.0.0.255
permit ip any any
exit
interface g0/0.20
ip access-group TPV-SOLO-ODOO in
exit

ip domain-name almichain.local
username almiadmin privilege 15 secret <contraseña>
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
! Solo se administra desde gestión (VLAN 10) y administración (VLAN 99): ni clientes, ni TPV, ni Internet
access-list 10 permit 192.168.10.0 0.0.0.255
access-list 10 permit 192.168.99.0 0.0.0.255
login block-for 120 attempts 3 within 60
line vty 0 15
access-class 10 in
login local
transport input ssh
exec-timeout 5 0
exit
line console 0
login local
exec-timeout 5 0
exit
end
write memory
